Quickstart

# validate input before spending a run
curl -H "Authorization: Bearer $TOKEN" \
     -H "Content-Type: application/json" \
     -d '{"input": {"url": "https://example.com", "strategy": "auto"}}' \
     {{ request.base_url }}api/v1/actors/universal-web/validate

# start a run
curl -X POST -H "Authorization: Bearer $TOKEN" \
     -H "Content-Type: application/json" \
     -d '{"input": {"url": "https://example.com", "strategy": "auto"}, "name": "my first run"}' \
     {{ request.base_url }}api/v1/actors/universal-web/runs

# poll the run, then read the dataset
curl -H "Authorization: Bearer $TOKEN" {{ request.base_url }}api/v1/runs/{run_id}
curl -H "Authorization: Bearer $TOKEN" {{ request.base_url }}api/v1/datasets/{dataset_id}/items

# export
curl -X POST -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
     -d '{"format": "csv"}' {{ request.base_url }}api/v1/datasets/{dataset_id}/export
